/* Estado en memoria (nunca localStorage): credenciales del Administrador y cuentas
   cargadas, solo mientras esta pestaña está abierta. */
var ADMIN_USUARIO = null;
var ADMIN_PIN = null;
var CUENTAS = [];
var CUENTA_SELECCIONADA = null;

/* =====================================================================================
 * CONEXIÓN AL BACKEND — CORREGIDO 10 sep 2026: esta página YA NO se sirve desde
 * Apps Script (por eso la huella funciona: aquí no hay iframe de por medio). El
 * backend (Código.gs, en tu proyecto de Apps Script) ahora es una API de datos, y
 * esta página le habla con fetch() normal en vez de google.script.run.
 * PEGA AQUÍ la URL de tu implementación de Apps Script (termina en /exec):
 * ===================================================================================== */
var APPS_SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbzGqD22nBUZHRI1cpIrldlm8GycqUo3k8HVYcwptYB8ll40UtHFmdRR_wcNwcV-lGOB/exec';

function llamarServidor(nombreFn){
  var args = Array.prototype.slice.call(arguments, 1);
  return fetch(APPS_SCRIPT_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
    body: JSON.stringify({ fn: nombreFn, args: args })
  })
    .then(function(r){ return r.json(); })
    .then(function(resp){
      if (resp && resp.ok === false) { throw new Error(resp.error || 'Error desconocido.'); }
      return resp;
    })
    .catch(function(err){
      throw (err && err.message) ? err.message : String(err);
    });
}

function mostrarPantalla(id){
  document.querySelectorAll('.pantalla').forEach(function(el){ el.hidden = true; });
  document.getElementById(id).hidden = false;
  window.scrollTo(0,0);
}
function ponerError(idBanner, mensaje){
  var el = document.getElementById(idBanner);
  el.textContent = mensaje; el.hidden = false;
}
function limpiarError(idBanner){
  var el = document.getElementById(idBanner);
  el.hidden = true; el.textContent = '';
}
function conBotonCargando(btn, textoCargando, fn){
  var original = btn.textContent;
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner"></span>' + textoCargando;
  return fn().finally(function(){ btn.disabled = false; btn.textContent = original; });
}

/* ---- Login (solo Administrador — listarCuentasHuella rechaza a cualquier otro) ---- */
function hacerLogin(){
  limpiarError('errLogin');
  var usuario = document.getElementById('inUsuario').value.trim();
  var pin = document.getElementById('inPin').value.trim();
  if (!usuario || !pin) { ponerError('errLogin','Escribe usuario y PIN.'); return; }

  var btn = document.getElementById('btnLogin');
  conBotonCargando(btn, 'Entrando...', function(){
    return llamarServidor('listarCuentasHuella', usuario, pin).then(function(resp){
      ADMIN_USUARIO = usuario;
      ADMIN_PIN = pin;
      CUENTAS = resp.cuentas || [];
      document.getElementById('inPin').value = '';
      document.getElementById('btnSalir').hidden = false;
      pintarLista();
      mostrarPantalla('p-lista');
    }).catch(function(err){ ponerError('errLogin', err); });
  });
}

function cerrarSesion(){
  ADMIN_USUARIO = null; ADMIN_PIN = null; CUENTAS = []; CUENTA_SELECCIONADA = null;
  document.getElementById('btnSalir').hidden = true;
  document.getElementById('inUsuario').value = '';
  mostrarPantalla('p-login');
}

/* ---- Lista de cuentas ---- */
function cargarCuentas(){
  limpiarError('errLista');
  llamarServidor('listarCuentasHuella', ADMIN_USUARIO, ADMIN_PIN).then(function(resp){
    CUENTAS = resp.cuentas || [];
    pintarLista();
  }).catch(function(err){ ponerError('errLista', err); });
}

function pintarLista(){
  var cont = document.getElementById('listaCuentas');
  cont.innerHTML = '';
  if (!CUENTAS.length) {
    cont.innerHTML = '<p class="campo-ayuda">No hay cuentas todavía (agrega filas en USUARIOS_EQUIPO).</p>';
    return;
  }
  CUENTAS.forEach(function(c){
    var b = document.createElement('button');
    b.className = 'fila-cuenta';
    b.onclick = function(){ abrirRegistro(c); };
    var rol = c.esAdmin ? (c.rolDepartamento || 'Supervisor/Calidad') : (c.rolDepartamento || 'Equipo');
    b.innerHTML =
      '<div class="info"><b>' + (c.nombreLider || c.usuario) + '</b>' +
      '<span>' + c.equipoId + ' · ' + rol + '</span></div>' +
      '<span class="estado-huella ' + (c.tieneHuella ? 'estado-si' : 'estado-no') + '">' +
      (c.tieneHuella ? 'Con huella' : 'Sin huella') + '</span>';
    cont.appendChild(b);
  });
}

/* ---- Registrar / re-registrar huella ---- */
function abrirRegistro(cuenta){
  limpiarError('errRegistrar');
  document.getElementById('okRegistrar').hidden = true;
  CUENTA_SELECCIONADA = cuenta;
  document.getElementById('registrarTitulo').textContent = cuenta.tieneHuella ? 'Volver a registrar huella' : 'Registrar huella';
  document.getElementById('registrarSubt').textContent = cuenta.equipoId + ' · ' + (cuenta.rolDepartamento || '');
  document.getElementById('inNombreLider').value = cuenta.nombreLider || '';
  document.getElementById('registrarEstado').textContent = 'Escribe el nombre y toca el botón para capturar la huella.';
  document.getElementById('diagnosticoContexto').textContent =
    'Diagnóstico — dentro de iframe: ' + (window.self !== window.top ? 'SÍ' : 'no') +
    ' · origen: ' + location.origin +
    ' · WebAuthn disponible: ' + (window.PublicKeyCredential ? 'sí' : 'no');
  mostrarPantalla('p-registrar');
}

function bufferABase64url(buffer){
  var bytes = new Uint8Array(buffer);
  var binario = '';
  for (var i = 0; i < bytes.byteLength; i++) binario += String.fromCharCode(bytes[i]);
  var b64 = btoa(binario);
  return b64.replace(/\+/g,'-').replace(/\//g,'_').replace(/=+$/,'');
}

function capturarHuella(){
  limpiarError('errRegistrar');
  document.getElementById('okRegistrar').hidden = true;
  var nombreLider = document.getElementById('inNombreLider').value.trim();
  if (!nombreLider) { ponerError('errRegistrar','Escribe el nombre de quien va a usar esta huella.'); return; }

  if (!window.PublicKeyCredential) {
    ponerError('errRegistrar','Este navegador no soporta huella (WebAuthn). Usa el celular/tablet de la persona con Chrome o Safari actualizado.');
    return;
  }

  var equipoId = CUENTA_SELECCIONADA.equipoId;
  var estado = document.getElementById('registrarEstado');
  var btn = document.getElementById('btnRegistrarHuella');

  conBotonCargando(btn, 'Capturando...', function(){
    return llamarServidor('iniciarRegistroLider', equipoId).then(function(resp){
      estado.textContent = 'Sigue las instrucciones del dispositivo (huella/Face ID)...';
      var opciones = {
        publicKey: {
          challenge: new TextEncoder().encode(resp.challenge),
          rp: { name: 'VUHL — Escaneo de Piso' }, // sin "id": dejamos que el navegador use el origen real de la página, en vez de adivinarlo nosotros
          user: {
            id: new TextEncoder().encode(equipoId),
            name: equipoId,
            displayName: nombreLider
          },
          pubKeyCredParams: [{ type:'public-key', alg:-7 }, { type:'public-key', alg:-257 }],
          authenticatorSelection: { authenticatorAttachment:'platform', userVerification:'required', residentKey:'preferred' },
          timeout: 60000,
          attestation: 'none'
        }
      };
      return navigator.credentials.create(opciones).then(function(credencial){
        var pubKeyBuf = (credencial.response.getPublicKey && credencial.response.getPublicKey()) || null;
        return llamarServidor('registrarLiderEquipo', {
          equipoId: equipoId,
          nombreLider: nombreLider,
          credencial: {
            credentialId: credencial.id,
            publicKeyCoseB64: pubKeyBuf ? bufferABase64url(pubKeyBuf) : '',
            challenge: resp.challenge
          },
          idUsuarioAdmin: ADMIN_USUARIO,
          pinAdmin: ADMIN_PIN
        });
      }).then(function(){
        document.getElementById('okRegistrar').hidden = false;
        document.getElementById('okRegistrar').textContent = 'Huella registrada para ' + nombreLider + ' (' + equipoId + ').';
        estado.textContent = 'Listo. Puedes volver a la lista o registrar otra huella en este mismo dispositivo.';
        cargarCuentas();
      }).catch(function(errWebauthn){
        ponerError('errRegistrar', 'No se pudo capturar la huella (' +
          (errWebauthn && errWebauthn.name ? errWebauthn.name : 'error') + '). Vuelve a intentar en este mismo dispositivo.');
      });
    }).catch(function(err){ ponerError('errRegistrar', err); });
  });
}
